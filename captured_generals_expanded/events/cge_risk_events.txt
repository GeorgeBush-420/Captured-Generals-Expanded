add_namespace = cge_risk
add_namespace = cge_debug
country_event = {
    id = cge_risk.1
    title = CGE_RISK_SUICIDE_TITLE
    desc = CGE_RISK_SUICIDE_DESC
    picture = GFX_cge_risk_death
    is_triggered_only = yes
    option = {
        name = CGE_RISK_SUICIDE_ACK
        custom_effect_tooltip = CGE_RISK_LOSS_TT
        hidden_effect = {
            if = {
                limit = {
                    NOT = {
                        has_global_flag = cge_logic_disabled
                    }
                }
                hidden_effect = {
                    clr_country_flag = cge_risk_notice_pending
                }
            }
        }
    }
    immediate = {
        if = {
            limit = {
                NOT = {
                    has_global_flag = cge_logic_disabled
                }
            }
        }
        else = {
            set_country_flag = cge_profile_deferred_cge_risk_1
        }
    }
}
