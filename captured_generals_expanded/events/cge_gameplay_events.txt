unit_leader_event = {
    id = cge_recruitment.1
    title = CGE_RECRUITMENT_EVENT_TITLE
    desc = CGE_RECRUITMENT_EVENT_DESC
    is_triggered_only = yes
    immediate = {
        if = {
            limit = {
                NOT = {
                    has_variable = cge_cge_recruitment_date
                }
            }
            set_variable = {
                var = cge_cge_recruitment_date
                value = global.date
            }
            set_character_flag = cge_recruitment_event_date_fallback
        }
    }
    option = {
        name = CGE_RECRUITMENT_EVENT_OPTION
        ai_chance = {
            factor = 1
        }
        custom_effect_tooltip = CGE_RECRUITMENT_EVENT_OPTION_TT
        hidden_effect = {
            FROM = {
                add_war_support = 0.05
                if = {
                    limit = {
                        has_government = democratic
                    }
                    add_popularity = {
                        ideology = democratic
                        popularity = 0.05
                    }
                }
                else_if = {
                    limit = {
                        has_government = fascism
                    }
                    add_popularity = {
                        ideology = fascism
                        popularity = 0.05
                    }
                }
                else_if = {
                    limit = {
                        has_government = communism
                    }
                    add_popularity = {
                        ideology = communism
                        popularity = 0.05
                    }
                }
                else = {
                    add_popularity = {
                        ideology = neutrality
                        popularity = 0.05
                    }
                }
            }
            if = {
                limit = {
                    has_character_flag = cge_recruitment_event_date_fallback
                }
                clear_variable = cge_cge_recruitment_date
                clr_character_flag = cge_recruitment_event_date_fallback
            }
        }
    }
}
