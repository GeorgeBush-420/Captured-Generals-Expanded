add_namespace = cge_postwar
country_event = {
    id = cge_postwar.1
    hidden = yes
    is_triggered_only = yes
    immediate = {
        if = {
            limit = {
                NOT = {
                    has_global_flag = cge_logic_disabled
                }
            }
            if = {
                limit = {
                    check_variable = {
                        var = cge_postwar_origin_queue^num
                        value = 0
                        compare = greater_than
                    }
                }
                var:cge_postwar_origin_queue^0 = {
                    save_global_event_target_as = cge_postwar_origin
                }
                remove_from_array = {
                    array = cge_postwar_origin_queue
                    index = 0
                }
                cge_postwar_begin_origin = yes
            }
            else = {
                clear_array = cge_postwar_origin_queue
                clr_country_flag = cge_postwar_queue_active
            }
        }
        else = {
            set_country_flag = cge_profile_deferred_cge_postwar_1
        }
    }
}
