
on_actions = {

	on_peaceconference_ended = {

		effect = {


			if = {
				limit = {
					ROOT = { is_ai = no }
				}

				FROM = {
					save_global_event_target_as = cge_postwar_detected_origin
				}

				ROOT = {

					cge_postwar_arm_exact_peace_origin = yes

					if = {
						limit = {
							has_country_flag = cge_postwar_exact_origin_matched
						}

						clr_country_flag = cge_postwar_exact_origin_matched

						log = "CGE_POSTWAR peaceconference armed [event_target:cge_postwar_detected_origin.GetName] for [ROOT.GetName]."

						cge_postwar_schedule_event_scan = yes
					}
				}
			}



			if = {
				limit = {
					FROM = { is_ai = no }
				}

				ROOT = {
					save_global_event_target_as = cge_postwar_detected_origin
				}

				FROM = {

					cge_postwar_arm_exact_peace_origin = yes

					if = {
						limit = {
							has_country_flag = cge_postwar_exact_origin_matched
						}

						clr_country_flag = cge_postwar_exact_origin_matched

						log = "CGE_POSTWAR peaceconference armed [event_target:cge_postwar_detected_origin.GetName] for defeated player."

						cge_postwar_schedule_event_scan = yes
					}
				}
			}
		}
	}



	on_peace = {

		effect = {

			every_country = {
				limit = {
					is_ai = no
					has_country_flag = cge_postwar_watch_active
				}

				cge_postwar_schedule_event_scan = yes
			}
		}
	}
}
