on_actions = {
    on_peaceconference_ended = {
        effect = {
            if = {
                limit = {
                    NOT = {
                        has_global_flag = cge_logic_disabled
                    }
                }
                if = {
                    limit = {
                        ROOT = {
                            is_ai = no
                        }
                    }
                    FROM = {
                        save_global_event_target_as = cge_postwar_detected_origin
                    }
                    ROOT = {
                        cge_postwar_arm_exact_peace_origin = yes
                        if = {
                            limit = {
                                has_country_flag = cge_postwar_exact_origin_matched
                            }
                            clr_country_flag = cge_postwar_exact_origin_matched
                            cge_postwar_schedule_event_scan = yes
                        }
                    }
                }
                if = {
                    limit = {
                        FROM = {
                            is_ai = no
                        }
                    }
                    ROOT = {
                        save_global_event_target_as = cge_postwar_detected_origin
                    }
                    FROM = {
                        cge_postwar_arm_exact_peace_origin = yes
                        if = {
                            limit = {
                                has_country_flag = cge_postwar_exact_origin_matched
                            }
                            clr_country_flag = cge_postwar_exact_origin_matched
                            cge_postwar_schedule_event_scan = yes
                        }
                    }
                }
            }
        }
    }
    on_peace = {
        effect = {
            if = {
                limit = {
                    NOT = {
                        has_global_flag = cge_logic_disabled
                    }
                }
                every_country = {
                    limit = {
                        is_ai = no
                        has_country_flag = cge_postwar_watch_active
                    }
                    cge_postwar_schedule_event_scan = yes
                }
            }
        }
    }
    on_civil_war_end = {
        effect = {
            if = {
                limit = {
                    NOT = {
                        has_global_flag = cge_logic_disabled
                    }
                }
                every_country = {
                    limit = {
                        is_ai = no
                        check_variable = {
                            var = cge_captured_general_count
                            value = 0
                            compare = greater_than
                        }
                    }
                    cge_postwar_schedule_event_scan = yes
                }
            }
        }
    }
    on_annex = {
        effect = {
            if = {
                limit = {
                    NOT = {
                        has_global_flag = cge_logic_disabled
                    }
                }
                every_country = {
                    limit = {
                        is_ai = no
                        check_variable = {
                            var = cge_captured_general_count
                            value = 0
                            compare = greater_than
                        }
                    }
                    cge_postwar_schedule_event_scan = yes
                }
            }
        }
    }
}
