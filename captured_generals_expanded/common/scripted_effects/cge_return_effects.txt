cge_resolve_return_country = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        save_event_target_as = cge_return_captor
        clr_country_flag = cge_return_country_found
        clr_country_flag = cge_return_country_is_subject
        set_variable = {
            cge_return_best_score = -1
        }
        event_target:cge_return_origin = {
            save_event_target_as = cge_return_destination
        }
        cge_return_follow_successors = yes
        if = {
            limit = {
                NOT = {
                    has_country_flag = cge_return_country_found
                }
            }
            event_target:cge_return_origin = {
                var:original_tag = {
                    if = {
                        limit = {
                            any_owned_state = {
                                always = yes
                            }
                            NOT = {
                                has_war_with = event_target:cge_return_captor
                            }
                            NOT = {
                                tag = event_target:cge_return_captor
                            }
                        }
                        save_event_target_as = cge_return_destination
                        event_target:cge_return_captor = {
                            set_country_flag = cge_return_country_found
                        }
                    }
                }
            }
        }
        if = {
            limit = {
                NOT = {
                    has_country_flag = cge_return_country_found
                }
            }
            every_country = {
                limit = {
                    any_owned_state = {
                        always = yes
                    }
                    NOT = {
                        has_war_with = event_target:cge_return_captor
                    }
                    check_variable = {
                        var = original_tag
                        value = event_target:cge_return_origin.original_tag
                        compare = equals
                    }
                }
                set_variable = {
                    var = cge_return_candidate_score
                    value = num_owned_states
                }
                if = {
                    limit = {
                        is_subject_of = event_target:cge_return_captor
                    }
                    add_to_variable = {
                        cge_return_candidate_score = 10000
                    }
                }
                if = {
                    limit = {
                        check_variable = {
                            var = cge_return_candidate_score
                            value = event_target:cge_return_captor.cge_return_best_score
                            compare = greater_than
                        }
                    }
                    save_event_target_as = cge_return_destination
                    event_target:cge_return_captor = {
                        set_variable = {
                            var = cge_return_best_score
                            value = PREV.cge_return_candidate_score
                        }
                        set_country_flag = cge_return_country_found
                    }
                }
                clear_variable = cge_return_candidate_score
            }
        }
        if = {
            limit = {
                NOT = {
                    has_country_flag = cge_return_country_found
                }
            }
            every_country = {
                limit = {
                    is_subject_of = event_target:cge_return_captor
                    any_owned_state = {
                        is_core_of = event_target:cge_return_origin
                    }
                    NOT = {
                        has_war_with = event_target:cge_return_captor
                    }
                    event_target:cge_return_captor = {
                        NOT = {
                            has_country_flag = cge_return_country_found
                        }
                    }
                }
                save_event_target_as = cge_return_destination
                event_target:cge_return_captor = {
                    set_country_flag = cge_return_country_found
                }
            }
        }
        if = {
            limit = {
                event_target:cge_return_destination = {
                    is_subject_of = event_target:cge_return_captor
                }
            }
            set_country_flag = cge_return_country_is_subject
        }
        clear_variable = cge_return_best_score
    }
}
