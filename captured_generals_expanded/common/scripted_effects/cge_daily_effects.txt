    if = {
        limit = {
            NOT = {

            }
        }
        meta_effect = {
            text = {
                if = {
                    limit = {
                        has_country_flag = cge_has_candidate_[CGE_SLOT]
                        OR = {
                            check_variable = {
                                var = cge_slot_[CGE_SLOT]_executed
                                value = 0
                                compare = greater_than
                            }
                            check_variable = {
                                var = cge_slot_[CGE_SLOT]_recruited
                                value = 0
                                compare = greater_than
                            }
                        }
                    }
                    subtract_from_variable = {
                        cge_alive_count_work = 1
                    }
                }
            }
            CGE_SLOT = "[?cge_daily_slot|.0]"
        }
    }
}

cge_slot_apply_execution = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        meta_effect = {
            text = {
                if = {
                    limit = {
                        check_variable = {
                            var = cge_slot_[CGE_SLOT]_executed
                            value = 0
                            compare = greater_than
                        }
                        NOT = {
                            check_variable = {
                                var = cge_slot_[CGE_SLOT]_suicide
                                value = 0
                                compare = greater_than
                            }
                        }
                    }
                    set_variable = {
                        cge_slot_ctx = [CGE_SLOT]
                    }
                    cge_execute_selected_general = yes
                    clear_variable = cge_slot_ctx
                }
            }
            CGE_SLOT = "[?cge_daily_slot|.0]"
        }
    }
}
