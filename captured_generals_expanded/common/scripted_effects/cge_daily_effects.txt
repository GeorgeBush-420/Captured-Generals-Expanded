cge_daily_update_captured_generals = {
	set_variable = { var = cge_alive_captured_general_count value = cge_captured_general_count }
	clear_variable = cge_status_refresh_pending

	if = {
		limit = { has_variable = cge_overview_selected_slot }
		save_event_target_as = cge_execution_repair_captor
		set_variable = { cge_cooperation_consumer_goods_factor = 0 }
		set_variable = { cge_patience_interrogation_pp_factor = 0 }
		set_variable = { cge_patience_trial_consumer_goods_factor = 0 }
	}

	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_loop_slot
		set_variable = { var = cge_daily_slot value = cge_loop_slot }

		cge_slot_patience_decay = yes

		if = {
			limit = { has_variable = cge_overview_selected_slot }
			cge_slot_tick_operations = yes
			cge_slot_tick_trial = yes
			cge_slot_finish_eased = yes
			cge_slot_repair_execution = yes
			cge_slot_finish_execution = yes
			cge_slot_finish_patience_op = yes
			cge_slot_finish_cooperation_op = yes
			cge_slot_apply_recruitment = yes
			cge_slot_calc_consumer_penalty = yes
			cge_slot_calc_patience_modifiers = yes
			cge_slot_apply_execution = yes
		}

		# Count after status-changing operations so recruitment/execution is reflected immediately.
		cge_slot_count_alive = yes
	}

	if = {
		limit = { check_variable = { var = cge_alive_captured_general_count value = 0 compare = less_than } }
		set_variable = { cge_alive_captured_general_count = 0 }
	}
	clear_variable = cge_daily_slot

	if = {
		limit = { has_variable = cge_overview_selected_slot }
		cge_update_selected_progress = yes

		if = {
			limit = { check_variable = { var = cge_cooperation_consumer_goods_factor value = 0 compare = greater_than } }
			if = { limit = { NOT = { has_dynamic_modifier = { modifier = cge_cooperation_consumer_goods_penalty } } } add_dynamic_modifier = { modifier = cge_cooperation_consumer_goods_penalty } }
			force_update_dynamic_modifier = yes
		}
		else_if = { limit = { has_dynamic_modifier = { modifier = cge_cooperation_consumer_goods_penalty } } remove_dynamic_modifier = { modifier = cge_cooperation_consumer_goods_penalty } }

		if = {
			limit = { check_variable = { var = cge_patience_interrogation_pp_factor value = 0 compare = less_than } }
			if = { limit = { NOT = { has_dynamic_modifier = { modifier = cge_patience_interrogation_pp_penalty } } } add_dynamic_modifier = { modifier = cge_patience_interrogation_pp_penalty } }
			force_update_dynamic_modifier = yes
		}
		else_if = { limit = { has_dynamic_modifier = { modifier = cge_patience_interrogation_pp_penalty } } remove_dynamic_modifier = { modifier = cge_patience_interrogation_pp_penalty } }

		if = {
			limit = { check_variable = { var = cge_patience_trial_consumer_goods_factor value = 0 compare = greater_than } }
			if = { limit = { NOT = { has_dynamic_modifier = { modifier = cge_patience_trial_consumer_goods_penalty } } } add_dynamic_modifier = { modifier = cge_patience_trial_consumer_goods_penalty } }
			force_update_dynamic_modifier = yes
		}
		else_if = { limit = { has_dynamic_modifier = { modifier = cge_patience_trial_consumer_goods_penalty } } remove_dynamic_modifier = { modifier = cge_patience_trial_consumer_goods_penalty } }

		if = {
			limit = { check_variable = { var = cge_status_refresh_pending value = 0 compare = greater_than } }
			clear_variable = cge_status_refresh_pending
			cge_refresh_list_after_change = yes
			log = "CGE list refreshed after daily status change."
		}
	}
}


cge_slot_count_alive = {
	meta_effect = {
		text = {
			if = {
					limit = {
						has_country_flag = cge_has_candidate_[CGE_SLOT]
						OR = {
							check_variable = { var = cge_slot_[CGE_SLOT]_executed value = 0 compare = greater_than }
							event_target:cge_slot_[CGE_SLOT]_runtime_leader = { has_character_flag = cge_recruited_by_player }
						}
					}
					subtract_from_variable = { cge_alive_captured_general_count = 1 }
				}
		}
		CGE_SLOT = "[?cge_daily_slot|.0]"
	}
}

cge_calc_alive_count = {
	set_variable = { var = cge_alive_captured_general_count value = cge_captured_general_count }
	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_loop_slot

		set_variable = {
			var = cge_daily_slot
			value = cge_loop_slot
		}

		cge_slot_count_alive = yes
	}

	if = {
			limit = { check_variable = { var = cge_alive_captured_general_count value = 0 compare = less_than } }
			set_variable = { cge_alive_captured_general_count = 0 }
		}
	clear_variable = cge_daily_slot
}
