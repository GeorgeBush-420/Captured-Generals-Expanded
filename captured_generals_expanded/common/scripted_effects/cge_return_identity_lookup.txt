cge_return_bind_current_native = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        set_variable = {
            var = cge_cge_character_key
            value = event_target:cge_return_captor.cge_return_identity_key
        }
        set_variable = {
            var = cge_registry_character_key
            value = event_target:cge_return_captor.cge_return_identity_key
        }
        event_target:cge_return_captor = {
            if = {
                limit = {
                    check_variable = {
                        var = cge_return_native_bound
                        value = 0
                        compare = equals
                    }
                }
                add_to_array = {
                    array = cge_return_native_general
                    value = PREV
                }
                set_variable = {
                    cge_return_native_bound = 1
                }
            }
        }
    }
}
