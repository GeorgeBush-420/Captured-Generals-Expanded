cge_init_slots = {
	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_slot
		cge_init_slot = yes
	}
}

cge_rebuild_display_unsorted = {
	clear_array = cge_general_display_characters
	set_variable = { cge_visible_count = 0 }
	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_slot
		cge_init_slot = yes
		meta_effect = {
			text = {
				if = {
					limit = {
						has_country_flag = cge_has_candidate_[CGE_SLOT]
						OR = {
							has_country_flag = cge_show_executed_generals
							NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_executed value = 0 compare = greater_than } }
						}
					}
					add_to_array = { array = cge_general_display_characters value = event_target:cge_slot_[CGE_SLOT]_runtime_leader }
					add_to_variable = { cge_visible_count = 1 }
				}
			}
			CGE_SLOT = "[?cge_slot|.0]"
		}
	}
	set_variable = { var = cge_alive_captured_general_count value = cge_visible_count }
}

cge_load_candidate_sort_key = {
	meta_effect = {
		text = {
			if = { limit = { check_variable = { var = cge_sort_field value = 1 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_level } }
			else_if = { limit = { check_variable = { var = cge_sort_field value = 2 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_name_order } }
			else_if = { limit = { check_variable = { var = cge_sort_field value = 3 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_attack } }
			else_if = { limit = { check_variable = { var = cge_sort_field value = 4 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_defense } }
			else_if = { limit = { check_variable = { var = cge_sort_field value = 5 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_planning } }
			else_if = { limit = { check_variable = { var = cge_sort_field value = 6 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_logistics } }
			else_if = { limit = { check_variable = { var = cge_sort_field value = 7 compare = equals } } set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_capacity } }
			else = { set_variable = { var = cge_candidate_key value = cge_slot_[CGE_SLOT]_capture_order } }
			set_variable = { var = cge_candidate_order value = cge_slot_[CGE_SLOT]_capture_order }
		}
		CGE_SLOT = "[?cge_slot|.0]"
	}
}

cge_consider_sort_candidate = {
	meta_effect = {
		text = {
			if = {
				limit = {
					has_country_flag = cge_has_candidate_[CGE_SLOT]
					OR = { has_country_flag = cge_show_executed_generals NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_executed value = 0 compare = greater_than } } }
					event_target:cge_slot_[CGE_SLOT]_runtime_leader = { NOT = { check_variable = { var = cge_cge_sort_epoch value = ROOT.cge_sort_epoch compare = equals } } }
				}
				cge_load_candidate_sort_key = yes
				if = {
					limit = { check_variable = { var = cge_best_found value = 0 compare = equals } }
					set_variable = { cge_best_found = 1 }
					set_variable = { var = cge_best_key value = cge_candidate_key }
					set_variable = { var = cge_best_order value = cge_candidate_order }
					event_target:cge_slot_[CGE_SLOT]_runtime_leader = { save_event_target_as = cge_sort_best_character }
				}
				else_if = {
					limit = {
						OR = {
							AND = { check_variable = { var = cge_sort_desc value = 0 compare = greater_than } check_variable = { var = cge_candidate_key value = cge_best_key compare = greater_than } }
							AND = { NOT = { check_variable = { var = cge_sort_desc value = 0 compare = greater_than } } check_variable = { var = cge_candidate_key value = cge_best_key compare = less_than } }
							AND = { check_variable = { var = cge_candidate_key value = cge_best_key compare = equals } check_variable = { var = cge_candidate_order value = cge_best_order compare = less_than } }
						}
					}
					set_variable = { var = cge_best_key value = cge_candidate_key }
					set_variable = { var = cge_best_order value = cge_candidate_order }
					event_target:cge_slot_[CGE_SLOT]_runtime_leader = { save_event_target_as = cge_sort_best_character }
				}
			}
		}
		CGE_SLOT = "[?cge_slot|.0]"
	}
}

cge_sort_display = {
	clear_array = cge_general_display_characters
	add_to_variable = { cge_sort_epoch = 1 }
	set_variable = { cge_sort_passes = cge_captured_general_count }
	for_loop_effect = {
		start = 1
		end = cge_sort_passes
		compare = less_than_or_equals
		value = cge_sort_pass
		set_variable = { cge_best_found = 0 }
		for_loop_effect = {
			start = 1
			end = cge_captured_general_count
			compare = less_than_or_equals
			value = cge_slot
			cge_consider_sort_candidate = yes
		}
		if = {
			limit = { check_variable = { var = cge_best_found value = 0 compare = greater_than } }
			add_to_array = { array = cge_general_display_characters value = event_target:cge_sort_best_character }
			event_target:cge_sort_best_character = { set_variable = { var = cge_cge_sort_epoch value = ROOT.cge_sort_epoch } }
		}
	}
	set_variable = { var = cge_alive_captured_general_count value = cge_general_display_characters^num }
}

cge_cycle_sort_field = {
	if = {
		limit = { check_variable = { var = cge_sort_field value = cge_sort_requested_field compare = equals } }
		if = {
			limit = { check_variable = { var = cge_sort_desc value = 0 compare = greater_than } }
			set_variable = { cge_sort_desc = 0 }
		}
		else = {
			set_variable = { cge_sort_field = 8 }
			set_variable = { cge_sort_desc = 0 }
		}
	}
	else = {
		set_variable = { var = cge_sort_field value = cge_sort_requested_field }
		set_variable = { cge_sort_desc = 1 }
	}
	cge_sort_display = yes
	clear_variable = cge_sort_requested_field
}

cge_reapply_sort = {
	if = {
		limit = {
			has_variable = cge_sort_field
			check_variable = { var = cge_sort_field value = 0 compare = greater_than }
		}
		cge_sort_display = yes
	}
	else = {
		set_variable = { cge_sort_field = 8 }
		set_variable = { cge_sort_desc = 0 }
		cge_sort_display = yes
	}
}
