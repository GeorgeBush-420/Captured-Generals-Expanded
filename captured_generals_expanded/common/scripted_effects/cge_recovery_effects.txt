cge_recovery_resolve_recipient = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        save_event_target_as = cge_recovery_holder
        clr_country_flag = cge_recovery_recipient_found
        meta_effect = {
            text = {
                var:cge_slot_[SLOT]_runtime_leader = {
                    OWNER = {
                        if = {
                            limit = {
                                exists = yes
                                NOT = {
                                    tag = event_target:cge_recovery_holder
                                }
                            }
                            save_event_target_as = cge_recovery_recipient
                            event_target:cge_recovery_holder = {
                                set_country_flag = cge_recovery_recipient_found
                            }
                        }
                    }
                }
                var:cge_slot_[SLOT]_origin_country^0 = {
                    save_event_target_as = cge_recovery_original_home
                }
                every_country = {
                    limit = {
                        is_ai = no
                        exists = yes
                        NOT = {
                            tag = event_target:cge_recovery_holder
                        }
                        NOT = {
                            has_war_with = event_target:cge_recovery_holder
                        }
                        check_variable = {
                            var = original_tag
                            value = event_target:cge_recovery_original_home.original_tag
                            compare = equals
                        }
                    }
                    save_event_target_as = cge_recovery_recipient
                    event_target:cge_recovery_holder = {
                        set_country_flag = cge_recovery_recipient_found
                    }
                }
                if = {
                    limit = {
                        NOT = {
                            has_country_flag = cge_recovery_recipient_found
                        }
                    }
                    event_target:cge_recovery_original_home = {
                        save_event_target_as = cge_return_origin
                    }
                    cge_resolve_return_country = yes
                    if = {
                        limit = {
                            has_country_flag = cge_return_country_found
                        }
                        event_target:cge_return_destination = {
                            save_event_target_as = cge_recovery_recipient
                        }
                        set_country_flag = cge_recovery_recipient_found
                    }
                }
            }
            SLOT = "[?cge_recovery_offer_slot|.0]"
        }
    }
}
cge_recovery_offer_peacetime_prisoners = {

            }
        }
        save_event_target_as = cge_recovery_holder
        for_loop_effect = {
            start = 1
            end = cge_captured_general_count
            compare = less_than_or_equals
            value = cge_recovery_offer_index
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[SLOT]
                            has_variable = cge_slot_[SLOT]_runtime_leader
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_executed
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_recruited
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_pow_collaborator
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_recovery_state
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            var:cge_slot_[SLOT]_runtime_leader = {
                                OR = {
                                    is_general_captured = yes
                                    AND = {
                                        has_global_flag = cge_custody_settlement_pending
                                        has_variable = cge_native_captor
                                        check_variable = {
                                            var = cge_native_captor
                                            value = PREV
                                            compare = equals
                                        }
                                    }
                                }
                            }
                            var:cge_slot_[SLOT]_runtime_leader = {
                                OR = {
                                    NOT = {
                                        has_variable = cge_native_captor
                                    }
                                    check_variable = {
                                        var = cge_native_captor
                                        value = PREV
                                        compare = equals
                                    }
                                }
                            }
                            NOT = {
                                var:cge_slot_[SLOT]_held_state = {
                                    CONTROLLER = {
                                        exists = yes
                                        NOT = {
                                            tag = event_target:cge_recovery_holder
                                        }
                                        check_variable = {
                                            var = original_tag
                                            value = event_target:cge_recovery_holder.original_tag
                                            compare = equals
                                        }
                                    }
                                }
                            }
                        }
                        set_variable = {
                            cge_recovery_offer_slot = [SLOT]
                        }
                        cge_recovery_resolve_recipient = yes
                        if = {
                            limit = {
                                has_country_flag = cge_recovery_recipient_found
                                NOT = {
                                    has_war_with = event_target:cge_recovery_recipient
                                }
                                NOT = {
                                    is_in_array = {
                                        array = cge_custody_victors
                                        value = event_target:cge_recovery_recipient
                                    }
                                }
                                event_target:cge_recovery_recipient = {
                                    NOT = {
                                        tag = event_target:cge_recovery_holder
                                    }
                                }
                            }
                            set_variable = {
                                var = cge_slot_[SLOT]_recovery_recipient
                                value = event_target:cge_recovery_recipient
                            }
                            set_variable = {
                                cge_slot_[SLOT]_recovery_state = 1
                            }
                            var:cge_slot_[SLOT]_runtime_leader = {
                                if = {
                                    limit = {
                                        NOT = {
                                            is_general_captured = yes
                                        }
                                    }
                                    capture_by = event_target:cge_recovery_holder
                                }
                                set_variable = {
                                    var = cge_native_captor
                                    value = event_target:cge_recovery_holder
                                }
                            }
                            event_target:cge_recovery_recipient = {
                                if = {
                                    limit = {
                                        NOT = {
                                            is_in_array = {
                                                array = cge_recovery_queue
                                                value = event_target:cge_recovery_holder
                                            }
                                        }
                                    }
                                    add_to_array = {
                                        array = cge_recovery_queue
                                        value = event_target:cge_recovery_holder
                                    }
                                }
                                cge_recovery_schedule_next = yes
                            }
                        }
                    }
                }
                SLOT = "[?cge_recovery_offer_index|.0]"
            }
        }
    }
