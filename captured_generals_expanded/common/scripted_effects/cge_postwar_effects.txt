cge_postwar_prepare_event = {
	clear_array = cge_postwar_general_display
	clear_array = cge_postwar_origin_display
	clear_array = cge_postwar_destination_display

	if = {
		limit = { has_event_target = cge_postwar_general }
		add_to_array = {
			array = cge_postwar_general_display
			value = event_target:cge_postwar_general
		}
	}

	if = {
		limit = {
			NOT = { check_variable = { var = cge_postwar_general_display^num value = 0 compare = greater_than } }
			check_variable = { var = cge_postwar_selected_slot value = 0 compare = greater_than }
			check_variable = { var = cge_postwar_selected_slot value = cge_captured_general_count compare = less_than_or_equals }
		}
		meta_effect = {
			text = {
				if = {
					limit = { has_country_flag = cge_has_candidate_[CGE_SLOT] }
					add_to_array = {
						array = cge_postwar_general_display
						value = event_target:cge_slot_[CGE_SLOT]_runtime_leader
					}
				}
			}
			CGE_SLOT = "[?cge_postwar_selected_slot|.0]"
		}
	}

	if = {
		limit = {
			NOT = { check_variable = { var = cge_postwar_general_display^num value = 0 compare = greater_than } }
			check_variable = { var = cge_captured_general_count value = 0 compare = greater_than }
		}
		set_temp_variable = { cge_postwar_event_fallback_slot = 1 }
		meta_effect = {
			text = {
				if = {
					limit = { has_country_flag = cge_has_candidate_[CGE_SLOT] }
					add_to_array = {
						array = cge_postwar_general_display
						value = event_target:cge_slot_[CGE_SLOT]_runtime_leader
					}
				}
			}
			CGE_SLOT = "[?cge_postwar_event_fallback_slot|.0]"
		}
		clear_variable = cge_postwar_event_fallback_slot
	}

	if = {
		limit = {
			NOT = { check_variable = { var = cge_postwar_origin_display^num value = 0 compare = greater_than } }
			has_event_target = cge_postwar_origin
		}
		add_to_array = {
			array = cge_postwar_origin_display
			value = event_target:cge_postwar_origin
		}
	}

	if = {
		limit = {
			NOT = { check_variable = { var = cge_postwar_origin_display^num value = 0 compare = greater_than } }
			check_variable = { var = cge_postwar_selected_slot value = 0 compare = greater_than }
			check_variable = { var = cge_postwar_selected_slot value = cge_captured_general_count compare = less_than_or_equals }
		}
		meta_effect = {
			text = {
				if = {
					limit = {
						has_country_flag = cge_has_candidate_[CGE_SLOT]
						has_variable = cge_slot_[CGE_SLOT]_origin_country
					}
					var:cge_slot_[CGE_SLOT]_origin_country^0 = {
						save_event_target_as = cge_postwar_event_origin_display
					}
					add_to_array = {
						array = cge_postwar_origin_display
						value = event_target:cge_postwar_event_origin_display
					}
				}
			}
			CGE_SLOT = "[?cge_postwar_selected_slot|.0]"
		}
	}

	if = {
		limit = {
			NOT = { check_variable = { var = cge_postwar_origin_display^num value = 0 compare = greater_than } }
			check_variable = { var = cge_captured_general_count value = 0 compare = greater_than }
		}
		set_temp_variable = { cge_postwar_event_fallback_slot = 1 }
		meta_effect = {
			text = {
				if = {
					limit = {
						has_country_flag = cge_has_candidate_[CGE_SLOT]
						has_variable = cge_slot_[CGE_SLOT]_origin_country
					}
					var:cge_slot_[CGE_SLOT]_origin_country^0 = {
						save_event_target_as = cge_postwar_event_origin_display
					}
					add_to_array = {
						array = cge_postwar_origin_display
						value = event_target:cge_postwar_event_origin_display
					}
				}
			}
			CGE_SLOT = "[?cge_postwar_event_fallback_slot|.0]"
		}
		clear_variable = cge_postwar_event_fallback_slot
	}

	if = {
		limit = {
			NOT = { check_variable = { var = cge_postwar_destination_display^num value = 0 compare = greater_than } }
			has_event_target = cge_postwar_destination
		}
		add_to_array = {
			array = cge_postwar_destination_display
			value = event_target:cge_postwar_destination
		}
	}
}



cge_postwar_begin_origin = {

	cge_postwar_resolve_destination = yes

	cge_postwar_repatriate = yes

	if = {
		limit = {
			NOT = {
				has_country_flag = cge_postwar_ordinary_event_pending
			}
		}

		cge_postwar_begin_origin_core = yes
	}
}

cge_postwar_begin_origin_core = {

	clear_array = cge_postwar_origin_display

	add_to_array = {
		array = cge_postwar_origin_display
		value = event_target:cge_postwar_origin
	}





	event_target:cge_postwar_origin = {
		clr_country_flag = cge_postwar_governor_ready

		if = {
			limit = {
				exists = yes
				is_subject_of = ROOT
			}

			set_country_flag = cge_postwar_governor_ready

			log = "CGE_POSTWAR origin [THIS.GetName] is a direct subject of [ROOT.GetName]."
		}
	}


	every_country = {
		clr_country_flag = cge_postwar_current_origin
	}

	event_target:cge_postwar_origin = {
		set_country_flag = cge_postwar_current_origin
	}

	log = "CGE_POSTWAR processing origin=[event_target:cge_postwar_origin.GetName] captor=[ROOT.GetName]."

	log = "CGE_POSTWAR BEGIN origin=[event_target:cge_postwar_origin.GetName] captor=[ROOT.GetName]."
	clr_country_flag = cge_postwar_repatriated_any
	clr_country_flag = cge_postwar_waiting_for_general_decision
	clr_country_flag = cge_postwar_governor_not_first_page
	clr_country_flag = cge_postwar_governor_has_more
	clear_variable = cge_postwar_selected_slot
	clear_variable = cge_postwar_governor_cursor
	clear_variable = cge_postwar_candidate_slot_1
	clear_variable = cge_postwar_candidate_slot_2
	clear_variable = cge_postwar_candidate_slot_3
	clear_global_event_target = cge_postwar_general
	clear_global_event_target = cge_postwar_candidate_1
	clear_global_event_target = cge_postwar_candidate_2
	clear_global_event_target = cge_postwar_candidate_3

	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_slot_iter
		set_variable = { var = cge_slot_arg value = cge_slot_iter }
		cge_slot_clear_postwar_flag = yes
	}


	if = {
		limit = { event_target:cge_postwar_origin = { has_war_with = ROOT } }
		cge_postwar_finish_origin = yes
	}
	else_if = {
		limit = { event_target:cge_postwar_origin = { exists = yes } }
		cge_postwar_repatriate = yes
		if = {
			limit = { has_country_flag = cge_postwar_repatriated_any }
			country_event = { id = cge_postwar.2 }
		}
		else = {
			cge_postwar_continue_after_ordinary = yes
		}
	}
	else = {
		cge_postwar_scan_collaborators = yes
	}
}

cge_postwar_finish_origin = {

	var:cge_postwar_destination_display^0 = {
		save_global_event_target_as = cge_postwar_destination
	}

	if = {
		limit = {
			has_country_flag = cge_postwar_destination_is_subject

			NOT = {
				has_country_flag = cge_postwar_governor_considered
			}
		}

		set_country_flag = cge_postwar_governor_considered

		log = "CGE_POSTWAR entering governor path for resolved subject [event_target:cge_postwar_destination.GetName]."

		cge_postwar_offer_governor = yes
	}

	else = {

		cge_postwar_finish_origin_core = yes

		clr_country_flag = cge_postwar_destination_found
		clr_country_flag = cge_postwar_destination_is_subject
		clr_country_flag = cge_postwar_governor_considered
		clr_country_flag = cge_postwar_ordinary_event_pending

		clear_array = cge_postwar_destination_display

		clear_global_event_target = cge_postwar_destination
	}
}

	}

	clear_variable = cge_postwar_candidate_slot_1

	clear_global_event_target = cge_postwar_candidate_3
	clear_global_event_target = cge_postwar_origin

	}

cge_postwar_repatriate = {
	clr_country_flag = cge_postwar_ordinary_repatriated
	clr_country_flag = cge_postwar_repatriated_any
	clr_country_flag = cge_postwar_bulk_removed
	set_variable = { var = cge_slot value = cge_captured_general_count }
	while_loop_effect = {
		limit = { check_variable = { var = cge_slot value = 0 compare = greater_than } }
		set_variable = { cge_postwar_match = 0 }
		meta_effect = {
			text = {
				if = {
					limit = {
						has_country_flag = cge_has_candidate_[CGE_SLOT]
						var:cge_slot_[CGE_SLOT]_origin_country^0 = { has_country_flag = cge_postwar_current_origin }
						NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_executed value = 0 compare = greater_than } }
						NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_recruited value = 0 compare = greater_than } }
						NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_pow_collaborator value = 0 compare = greater_than } }
					}
					event_target:cge_slot_[CGE_SLOT]_runtime_leader = {
						if = { limit = { is_general_captured = yes } release_from_captivity = yes }
						set_nationality = event_target:cge_postwar_destination
						clr_character_flag = cge_captured_by_player
						clr_character_flag = cge_bribed_by_player
						clr_character_flag = cge_recruited_by_player
						clr_character_flag = cge_pow_recruitment_unlocked
					}
					set_variable = { cge_postwar_match = 1 }
				}
			}
			CGE_SLOT = "[?cge_slot|.0]"
		}
		if = {
			limit = { check_variable = { var = cge_postwar_match value = 0 compare = greater_than } }
			set_variable = { var = cge_release_slot_to_remove value = cge_slot }
			cge_release_remove_selected_slot = yes
			set_country_flag = cge_postwar_ordinary_repatriated
			set_country_flag = cge_postwar_repatriated_any
			set_country_flag = cge_postwar_bulk_removed
		}
		subtract_from_variable = { cge_slot = 1 }
	}
	if = { limit = { has_country_flag = cge_postwar_bulk_removed } cge_refresh_list_after_change = yes }
}


cge_postwar_scan_collaborators = {
	if = {
		limit = { NOT = { has_country_flag = cge_postwar_waiting_for_general_decision } }

		if = {
			limit = {
				has_country_flag = cge_postwar_destination_is_subject
				NOT = { has_country_flag = cge_postwar_governor_considered }
				NOT = { has_country_flag = cge_postwar_fate_phase_active }
			}
			log = "CGE_POSTWAR governor phase precedes recruited-general fate phase."
			cge_postwar_offer_governor = yes
		}
		else = {
			cge_postwar_try_autonomy_request = yes

			if = {
				limit = { NOT = { has_country_flag = cge_postwar_autonomy_request_fired } }
				cge_postwar_scan_recruited_fates = yes
			}

			clr_country_flag = cge_postwar_autonomy_request_fired
		}
	}
	else = {
		log = "CGE_POSTWAR scanner blocked: postwar general decision already pending."
	}
}

cge_postwar_try_autonomy_request = {
	clr_country_flag = cge_postwar_autonomy_request_fired
	clear_array = cge_postwar_autonomy_candidates

	if = {
		limit = {
			event_target:cge_postwar_origin = { NOT = { exists = yes } }
			any_owned_state = {
				is_core_of = event_target:cge_postwar_origin
				NOT = { is_core_of = ROOT }
				NOT = { is_claimed_by = ROOT }
			}
		}

		for_loop_effect = {
			start = 1
			end = cge_captured_general_count
			compare = less_than_or_equals
			value = cge_slot_iter
			set_variable = { var = cge_slot_arg value = cge_slot_iter }
			cge_slot_postwar_autonomy_candidate = yes
		}

		if = {
			limit = {
				check_variable = { var = cge_postwar_autonomy_candidates^num value = 0 compare = greater_than }
			}

			random_scope_in_array = {
				array = cge_postwar_autonomy_candidates
				save_global_event_target_as = cge_postwar_general

				ROOT = {
					set_variable = {
						var = cge_postwar_selected_slot
						value = event_target:cge_postwar_general.cge_cge_slot_index
					}
					set_country_flag = cge_postwar_waiting_for_general_decision
					set_country_flag = cge_postwar_autonomy_request_fired
					log = "CGE_POSTWAR_AUTONOMY randomly selected [event_target:cge_postwar_general.GetName] to request restoration of [event_target:cge_postwar_origin.GetName]."
					country_event = { id = cge_postwar.4 hours = 1 }
				}
			}
		}
	}

	clear_array = cge_postwar_autonomy_candidates
}

cge_postwar_scan_recruited_fates = {
	clr_country_flag = cge_postwar_fate_found

	if = {
		limit = { NOT = { has_country_flag = cge_postwar_fate_phase_active } }
		set_country_flag = cge_postwar_fate_phase_active
		for_loop_effect = {
			start = 1
			end = cge_captured_general_count
			compare = less_than_or_equals
			value = cge_slot_iter
			set_variable = { var = cge_slot_arg value = cge_slot_iter }
			cge_slot_postwar_init = yes
		}
		log = "CGE_POSTWAR_FATE initialized recruited-general fate phase."
	}

	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_slot_iter
		set_variable = { var = cge_slot_arg value = cge_slot_iter }
		cge_slot_postwar_fate = yes
	}

	if = {
		limit = { NOT = { has_country_flag = cge_postwar_fate_found } }
		clr_country_flag = cge_postwar_fate_phase_active
		clr_country_flag = cge_postwar_waiting_for_general_decision
		for_loop_effect = {
			start = 1
			end = cge_captured_general_count
			compare = less_than_or_equals
			value = cge_slot_iter
			set_variable = { var = cge_slot_arg value = cge_slot_iter }
			cge_slot_postwar_cleanup = yes
		}
		log = "CGE_POSTWAR_FATE all recruited generals processed; continuing postwar flow."
		cge_postwar_finish_origin = yes
	}
	clr_country_flag = cge_postwar_fate_found
}


cge_postwar_send_home = {

	event_target:cge_postwar_general = {
		clr_character_flag = cge_postwar_fate_processed
	}


	var:cge_postwar_destination_display^0 = {
		save_global_event_target_as = cge_postwar_destination
	}


	event_target:cge_postwar_general = {
		set_character_flag = cge_postwar_repatriate_remove
	}



	event_target:cge_postwar_general = {

		if = {
			limit = { is_general_captured = yes }
			release_from_captivity = yes
		}

		set_nationality = event_target:cge_postwar_destination

		clr_character_flag = cge_captured_by_player
		clr_character_flag = cge_bribed_by_player
		clr_character_flag = cge_recruited_by_player
		clr_character_flag = cge_pow_recruitment_unlocked
		clr_character_flag = cge_postwar_processed_this_settlement
	}


	clr_country_flag = cge_postwar_waiting_for_general_decision


	cge_postwar_remove_repatriated = yes

	event_target:cge_postwar_general = {
		clr_character_flag = cge_postwar_repatriate_remove
	}

	clear_global_event_target = cge_postwar_general

	cge_postwar_scan_collaborators = yes
}

cge_postwar_reimprison = {
	var:cge_postwar_destination_display^0 = {
		save_global_event_target_as = cge_postwar_destination
	}

	meta_effect = {
		text = {
			if = {
				limit = { check_variable = { var = cge_slot_[SLOT]_recruited value = 0 compare = greater_than } }
				event_target:cge_postwar_general = {
					set_nationality = event_target:cge_postwar_destination
					if = { limit = { NOT = { is_general_captured = yes } } capture_by = ROOT }
				}
			}
			event_target:cge_postwar_general = {
				clr_character_flag = cge_postwar_processed_this_settlement
				clr_character_flag = cge_recruited_by_player
				clr_character_flag = cge_pow_recruitment_unlocked
				set_character_flag = cge_captured_by_player
			}
			if = {
				limit = {
					check_variable = { var = cge_slot_[SLOT]_pow_collaborator value = 0 compare = greater_than }
					has_variable = cge_slot_[SLOT]_origin_country
				}
				save_event_target_as = cge_postwar_captor_scope
				var:cge_slot_[SLOT]_origin_country^0 = {
					if = {
						limit = {
							check_variable = {
								var = cge_pow_recruiter_character_key
								value = event_target:cge_postwar_captor_scope.cge_slot_[SLOT]_character_key
								compare = equals
							}
						}
						cge_pow_clear_origin_recruiter_effect = yes
					}
				}
			}

			set_variable = { cge_slot_[SLOT]_recruited = 0 }
			set_variable = { cge_slot_[SLOT]_pow_collaborator = 0 }
			clear_array = cge_slot_[SLOT]_captor_country_display
			add_to_array = { array = cge_slot_[SLOT]_captor_country_display value = ROOT }
			add_to_variable = { cge_main_window_dirty = 1 }
		}
		SLOT = "[?cge_postwar_selected_slot|.0]"
	}
	clr_country_flag = cge_postwar_waiting_for_general_decision


	cge_postwar_refresh_list = yes



	cge_refresh_list_after_change = yes

}


cge_postwar_restore_origin_as_puppet = {
	every_owned_state = {
		limit = {
			is_core_of = event_target:cge_postwar_origin
			OR = { is_core_of = ROOT is_claimed_by = ROOT }
		}
		remove_core_of = event_target:cge_postwar_origin
		set_state_flag = cge_postwar_restore_origin_core_after_release
	}

	release_puppet = event_target:cge_postwar_origin

	# release_puppet creates the restored homeland as our subject.  Re-bind the
	# destination immediately so the next collaborator scan enters the governor path.
	event_target:cge_postwar_origin = {
		save_global_event_target_as = cge_postwar_destination
		if = {
			limit = { exists = yes }
			set_country_flag = cge_postwar_governor_ready
			log = "CGE_POSTWAR restored homeland [THIS.GetName] marked for military-governor selection."
		}
	}

	set_country_flag = cge_postwar_destination_found
	set_country_flag = cge_postwar_destination_is_subject
	clr_country_flag = cge_postwar_governor_considered
	clear_array = cge_postwar_destination_display
	add_to_array = { array = cge_postwar_destination_display value = event_target:cge_postwar_destination }
	log = "CGE_POSTWAR restored destination rebound to subject [event_target:cge_postwar_destination.GetName]; governor path armed."

	every_state = {
		limit = { has_state_flag = cge_postwar_restore_origin_core_after_release }
		add_core_of = event_target:cge_postwar_origin
		clr_state_flag = cge_postwar_restore_origin_core_after_release
	}

	clr_country_flag = cge_postwar_waiting_for_general_decision

	clr_country_flag = cge_postwar_repatriated_any
	cge_postwar_repatriate = yes
	if = {
		limit = { has_country_flag = cge_postwar_repatriated_any }
		country_event = { id = cge_postwar.2 }
	}
	else = { cge_postwar_continue_after_ordinary = yes }
}


cge_postwar_offer_governor = {
	set_country_flag = cge_postwar_governor_considered
	clr_country_flag = cge_postwar_has_governor_candidate

	for_loop_effect = {
		start = 1
		end = cge_captured_general_count
		compare = less_than_or_equals
		value = cge_slot_iter
		set_variable = { var = cge_slot_arg value = cge_slot_iter }
		cge_slot_governor_probe = yes
	}

				promote_leader = yes
