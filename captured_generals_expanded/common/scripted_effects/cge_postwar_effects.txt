cge_postwar_prepare_event = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        clear_array = cge_postwar_general_display
        clear_array = cge_postwar_origin_display
        clear_array = cge_postwar_destination_display
        if = {
            limit = {
                has_event_target = cge_postwar_general
            }
            add_to_array = {
                array = cge_postwar_general_display
                value = event_target:cge_postwar_general
            }
        }
        if = {
            limit = {
                NOT = {
                    check_variable = {
                        var = cge_postwar_general_display^num
                        value = 0
                        compare = greater_than
                    }
                }
                check_variable = {
                    var = cge_postwar_selected_slot
                    value = 0
                    compare = greater_than
                }
                check_variable = {
                    var = cge_postwar_selected_slot
                    value = cge_captured_general_count
                    compare = less_than_or_equals
                }
            }
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[CGE_SLOT]
                        }
                        add_to_array = {
                            array = cge_postwar_general_display
                            value = var:cge_slot_[CGE_SLOT]_runtime_leader
                        }
                    }
                }
                CGE_SLOT = "[?cge_postwar_selected_slot|.0]"
            }
        }
        if = {
            limit = {
                NOT = {
                    check_variable = {
                        var = cge_postwar_general_display^num
                        value = 0
                        compare = greater_than
                    }
                }
                check_variable = {
                    var = cge_captured_general_count
                    value = 0
                    compare = greater_than
                }
            }
            set_temp_variable = {
                cge_postwar_event_fallback_slot = 1
            }
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[CGE_SLOT]
                        }
                        add_to_array = {
                            array = cge_postwar_general_display
                            value = var:cge_slot_[CGE_SLOT]_runtime_leader
                        }
                    }
                }
                CGE_SLOT = "[?cge_postwar_event_fallback_slot|.0]"
            }
            clear_variable = cge_postwar_event_fallback_slot
        }
        if = {
            limit = {
                NOT = {
                    check_variable = {
                        var = cge_postwar_origin_display^num
                        value = 0
                        compare = greater_than
                    }
                }
                has_event_target = cge_postwar_origin
            }
            add_to_array = {
                array = cge_postwar_origin_display
                value = event_target:cge_postwar_origin
            }
        }
        if = {
            limit = {
                NOT = {
                    check_variable = {
                        var = cge_postwar_origin_display^num
                        value = 0
                        compare = greater_than
                    }
                }
                check_variable = {
                    var = cge_postwar_selected_slot
                    value = 0
                    compare = greater_than
                }
                check_variable = {
                    var = cge_postwar_selected_slot
                    value = cge_captured_general_count
                    compare = less_than_or_equals
                }
            }
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[CGE_SLOT]
                            has_variable = cge_slot_[CGE_SLOT]_origin_country
                        }
                        var:cge_slot_[CGE_SLOT]_origin_country^0 = {
                            save_event_target_as = cge_postwar_event_origin_display
                        }
                        add_to_array = {
                            array = cge_postwar_origin_display
                            value = event_target:cge_postwar_event_origin_display
                        }
                    }
                }
                CGE_SLOT = "[?cge_postwar_selected_slot|.0]"
            }
        }
        if = {
            limit = {
                NOT = {
                    check_variable = {
                        var = cge_postwar_origin_display^num
                        value = 0
                        compare = greater_than
                    }
                }
                check_variable = {
                    var = cge_captured_general_count
                    value = 0
                    compare = greater_than
                }
            }
            set_temp_variable = {
                cge_postwar_event_fallback_slot = 1
            }
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[CGE_SLOT]
                            has_variable = cge_slot_[CGE_SLOT]_origin_country
                        }
                        var:cge_slot_[CGE_SLOT]_origin_country^0 = {
                            save_event_target_as = cge_postwar_event_origin_display
                        }
                        add_to_array = {
                            array = cge_postwar_origin_display
                            value = event_target:cge_postwar_event_origin_display
                        }
                    }
                }
                CGE_SLOT = "[?cge_postwar_event_fallback_slot|.0]"
            }
            clear_variable = cge_postwar_event_fallback_slot
        }
        if = {
            limit = {
                NOT = {
                    check_variable = {
                        var = cge_postwar_destination_display^num
                        value = 0
                        compare = greater_than
                    }

            add_to_array = {
                array = cge_postwar_destination_display
                value = event_target:cge_postwar_destination
            }
        }
    }

cge_postwar_begin_origin = {
    if = {
        limit = {
            NOT = {

            }
        }
        cge_postwar_resolve_destination = yes
        cge_postwar_begin_origin_core = yes
    }
}
cge_postwar_begin_origin_core = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        clear_array = cge_postwar_origin_display
        add_to_array = {
            array = cge_postwar_origin_display
            value = event_target:cge_postwar_origin
        }
        event_target:cge_postwar_origin = {
            clr_country_flag = cge_postwar_governor_ready
            if = {
                limit = {
                    exists = yes
                    is_subject_of = ROOT
                }
                set_country_flag = cge_postwar_governor_ready
            }
        }
        cge_postwar_mark_origin_family = yes
        clr_country_flag = cge_postwar_repatriated_any
        clr_country_flag = cge_postwar_waiting_for_general_decision
        clr_country_flag = cge_postwar_governor_not_first_page
        clr_country_flag = cge_postwar_governor_has_more
        clear_variable = cge_postwar_selected_slot
        clear_variable = cge_postwar_governor_cursor
        clear_variable = cge_postwar_candidate_slot_1
        clear_variable = cge_postwar_candidate_slot_2
        clear_variable = cge_postwar_candidate_slot_3
        clear_global_event_target = cge_postwar_general
        clear_global_event_target = cge_postwar_candidate_1
        clear_global_event_target = cge_postwar_candidate_2
        clear_global_event_target = cge_postwar_candidate_3
        for_loop_effect = {
            start = 1
            end = cge_captured_general_count
            compare = less_than_or_equals
            value = cge_slot_iter
            set_variable = {
                var = cge_slot_arg
                value = cge_slot_iter
            }
            cge_slot_clear_postwar_flag = yes
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[CGE_SLOT]
                            var:cge_slot_[CGE_SLOT]_origin_country^0 = {
                                check_variable = {
                                    var = original_tag
                                    value = event_target:cge_postwar_origin.original_tag
                                    compare = equals
                                }
                            }
                        }
                        var:cge_slot_[CGE_SLOT]_runtime_leader = {
                            clr_character_flag = cge_postwar_war_pending
                        }
                    }
                }
                CGE_SLOT = "[?cge_slot_arg|.0]"
            }
        }
        if = {
            limit = {
                any_enemy_country = {
                    check_variable = {
                        var = original_tag
                        value = event_target:cge_postwar_origin.original_tag
                        compare = equals
                    }
                }
            }
            set_country_flag = cge_postwar_governor_considered
            cge_postwar_finish_origin = yes
        }
        else = {
            cge_postwar_repatriate = yes
            cge_postwar_continue_after_ordinary = yes
        }
    }
}
cge_postwar_finish_origin = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        var:cge_postwar_destination_display^0 = {
            save_global_event_target_as = cge_postwar_destination
        }
        if = {
            limit = {
                has_country_flag = cge_postwar_destination_is_subject
                NOT = {
                    has_country_flag = cge_postwar_governor_considered
                }
            }
            set_country_flag = cge_postwar_governor_considered
            cge_postwar_offer_governor = yes
        }
        else = {
            cge_postwar_finish_origin_core = yes
            clr_country_flag = cge_postwar_destination_found
            clr_country_flag = cge_postwar_destination_is_subject
            clr_country_flag = cge_postwar_governor_considered
            clr_country_flag = cge_postwar_ordinary_event_pending
            clear_array = cge_postwar_destination_display
            clear_global_event_target = cge_postwar_destination
        }
    }
}

                                has_country_flag = cge_postwar_origin_match

                                has_variable = cge_slot_[CGE_SLOT]_runtime_leader

                            }

            }
        }
        every_possible_country = {
            limit = {
                has_country_flag = cge_postwar_origin_match
            }
            clr_country_flag = cge_postwar_origin_match
        }
        clr_country_flag = cge_postwar_repatriated_any
        clr_country_flag = cge_postwar_waiting_for_general_decision
        clr_country_flag = cge_postwar_governor_not_first_page

        clear_variable = cge_postwar_candidate_slot_2
        clear_variable = cge_postwar_candidate_slot_3
        clear_global_event_target = cge_postwar_general
        clear_global_event_target = cge_postwar_candidate_1
        clear_global_event_target = cge_postwar_candidate_2
        clear_global_event_target = cge_postwar_candidate_3
        clear_global_event_target = cge_postwar_origin
        if = {
            limit = {
                check_variable = {
                    var = cge_postwar_origin_queue^num
                    value = 0
                    compare = greater_than
                }
            }
            country_event = {
                id = cge_postwar.1
                hours = 1
            }
        }
        else = {
            clear_array = cge_postwar_origin_queue
            clr_country_flag = cge_postwar_queue_active
        }
        if = {
            limit = {
                has_country_flag = cge_postwar_watch_active
            }
            cge_postwar_schedule_event_scan = yes
        }
    }
}

cge_postwar_repatriate_automatic = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        clr_country_flag = cge_postwar_ordinary_repatriated
        clr_country_flag = cge_postwar_repatriated_any
        set_variable = {
            cge_return_batch_pending_count = 0
        }
        for_loop_effect = {
            start = 1
            end = cge_captured_general_count
            compare = less_than_or_equals
            value = cge_repatriation_scan_index
            meta_effect = {
                text = {
                    clear_variable = cge_slot_[SLOT]_return_pending
                    if = {
                        limit = {
                            has_country_flag = cge_has_candidate_[SLOT]
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_negotiation_state
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            has_variable = cge_slot_[SLOT]_runtime_leader
                            var:cge_slot_[SLOT]_origin_country^0 = {
                                check_variable = {
                                    var = original_tag
                                    value = event_target:cge_postwar_origin.original_tag
                                    compare = equals
                                }
                            }
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_executed
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_recruited
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            NOT = {
                                check_variable = {
                                    var = cge_slot_[SLOT]_pow_collaborator
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            var:cge_slot_[SLOT]_runtime_leader = {
                                NOT = {
                                    has_character_flag = cge_recruited_by_player
                                }
                                NOT = {
                                    has_character_flag = cge_pow_recruitment_unlocked
                                }
                            }
                            NOT = {
                                any_enemy_country = {
                                    check_variable = {
                                        var = original_tag
                                        value = event_target:cge_postwar_origin.original_tag
                                        compare = equals
                                    }
                                }
                            }
                        }
                        set_variable = {
                            cge_slot_[SLOT]_return_pending = 1
                        }
                        add_to_variable = {
                            cge_return_batch_pending_count = 1
                        }
                    }
                }
                SLOT = "[?cge_repatriation_scan_index|.0]"
            }
        }
        if = {
            limit = {
                check_variable = {
                    var = cge_return_batch_pending_count
                    value = 0
                    compare = greater_than
                }
            }
            set_variable = {
                cge_return_batch_kind = 1
            }
            clear_array = cge_return_batch_origin
            add_to_array = {
                array = cge_return_batch_origin
                value = event_target:cge_postwar_origin
            }
            cge_return_begin_batch = yes
        }
        clear_variable = cge_return_batch_pending_count
    }
}
cge_postwar_scan_collaborators = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        if = {
            limit = {
                NOT = {
                    has_country_flag = cge_postwar_waiting_for_general_decision
                }
            }
            if = {
                limit = {
                    has_country_flag = cge_postwar_destination_is_subject
                    NOT = {
                        has_country_flag = cge_postwar_governor_considered
                    }
                    NOT = {
                        has_country_flag = cge_postwar_fate_phase_active
                    }
                }
                cge_postwar_offer_governor = yes
            }
            else = {
                cge_postwar_try_autonomy_request = yes
                if = {
                    limit = {
                        NOT = {
                            has_country_flag = cge_postwar_autonomy_request_fired
                        }
                    }
                    cge_postwar_scan_recruited_fates = yes
                }
                clr_country_flag = cge_postwar_autonomy_request_fired
            }
        }
        else = {
        }
    }
}
cge_postwar_try_autonomy_request = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        clr_country_flag = cge_postwar_autonomy_request_fired
        clear_array = cge_postwar_autonomy_candidates
        if = {
            limit = {
                event_target:cge_postwar_destination = {
                    NOT = {
                        exists = yes
                    }
                }
                any_owned_state = {
                    is_core_of = event_target:cge_postwar_origin
                    NOT = {
                        is_core_of = ROOT
                    }
                    NOT = {
                        is_claimed_by = ROOT
                    }
                }
            }
            for_loop_effect = {
                start = 1
                end = cge_captured_general_count
                compare = less_than_or_equals
                value = cge_slot_iter
                set_variable = {
                    var = cge_slot_arg
                    value = cge_slot_iter
                }
                cge_slot_postwar_autonomy_candidate = yes
            }
            if = {
                limit = {
                    check_variable = {
                        var = cge_postwar_autonomy_candidates^num
                        value = 0
                        compare = greater_than
                    }
                }
                random_scope_in_array = {
                    array = cge_postwar_autonomy_candidates
                    save_global_event_target_as = cge_postwar_general
                    ROOT = {
                        set_variable = {
                            var = cge_postwar_selected_slot
                            value = event_target:cge_postwar_general.cge_cge_slot_index
                        }
                        set_country_flag = cge_postwar_waiting_for_general_decision
                        set_country_flag = cge_postwar_autonomy_request_fired
                        country_event = {
                            id = cge_postwar.4
                            hours = 1
                        }
                    }
                }
            }
        }
        clear_array = cge_postwar_autonomy_candidates
    }
}
