
cge_trial_prepare_slot = {
	set_variable = { var = cge_slot_ctx value = event_target:cge_trial_general.cge_cge_slot_index }
}

cge_trial_finish_slot_refresh = {
	if = {
		limit = {
			check_variable = { var = cge_slot_ctx value = 0 compare = greater_than }
			check_variable = { var = cge_slot_ctx value = cge_captured_general_count compare = less_than_or_equals }
		}
		set_variable = { var = cge_slot value = cge_slot_ctx }
		cge_mirror_slot_to_character = yes
		clear_variable = cge_slot
	}
	clear_variable = cge_slot_ctx
}

cge_trial_stage1_public = {
	cge_trial_prepare_slot = yes
	meta_effect = {
		text = {
			if = {
				limit = { has_country_flag = cge_has_candidate_[CGE_SLOT] check_variable = { var = cge_slot_[CGE_SLOT]_patience_operation_pending value = 0 compare = greater_than } check_variable = { var = cge_slot_[CGE_SLOT]_patience_selected_action value = 5 compare = equals } NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_executed value = 0 compare = greater_than } } NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_recruited value = 0 compare = greater_than } } }
				set_variable = { cge_slot_[CGE_SLOT]_trial_route = 1 }
				set_variable = { cge_slot_[CGE_SLOT]_trial_publicity = 10 }
				set_variable = { cge_slot_[CGE_SLOT]_trial_stage = 1 }
			}
		}
		CGE_SLOT = "[?cge_slot_ctx|.0]"
	}
	cge_trial_finish_slot_refresh = yes
}
