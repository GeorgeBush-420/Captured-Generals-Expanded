cge_pow_add_digit = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        for_loop_effect = {
            start = 1
            end = 9
            compare = less_than_or_equals
            value = cge_pow_digit_iteration
            set_temp_variable = {
                var = cge_pow_test_thousands
                value = cge_pow_inflicted_thousands
            }
            add_to_temp_variable = {
                var = cge_pow_test_thousands
                value = cge_pow_digit_step
            }
            subtract_from_temp_variable = {
                cge_pow_test_thousands = 0.001
            }
            if = {
                limit = {
                    casualties_inflicted_by = {
                        opponent = event_target:cge_pow_captor
                        thousands > cge_pow_test_thousands
                    }
                }
                add_to_temp_variable = {
                    var = cge_pow_inflicted_thousands
                    value = cge_pow_digit_step
                }
            }
        }
    }
}

    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        set_variable = {
            var = cge_timer_frame_work
            value = cge_timer_duration_work
        }
        subtract_from_variable = {
            var = cge_timer_frame_work
            value = cge_timer_days_left_work
        }
        multiply_variable = {
            cge_timer_frame_work = 30
        }
        divide_variable = {
            var = cge_timer_frame_work
            value = cge_timer_duration_work
        }
        subtract_from_variable = {
            cge_timer_frame_work = 0.499
        }
        round_variable = cge_timer_frame_work
        multiply_variable = {
            cge_timer_frame_work = 100
        }
        divide_variable = {
            cge_timer_frame_work = 30
        }

            cge_timer_frame_work = 0.499
