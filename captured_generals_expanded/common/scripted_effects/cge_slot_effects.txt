cge_slot_postwar_watch = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        meta_effect = {
            text = {
                if = {
                    limit = {
                        has_country_flag = cge_has_candidate_[CGE_SLOT]
                        check_variable = {
                            var = cge_slot_[CGE_SLOT]_origin_country^num
                            value = 0
                            compare = greater_than
                        }
                        NOT = {
                            check_variable = {
                                var = cge_slot_[CGE_SLOT]_executed
                                value = 0
                                compare = greater_than
                            }
                        }
                    }
                    var:cge_slot_[CGE_SLOT]_origin_country^0 = {
                        var:original_tag = {
                            save_event_target_as = cge_postwar_watched_origin
                        }
                    }
                    if = {
                        limit = {
                            OR = {
                                var:cge_slot_[CGE_SLOT]_origin_country^0 = {
                                    has_country_flag = cge_postwar_was_at_war_with_player
                                }
                                event_target:cge_postwar_watched_origin = {
                                    has_country_flag = cge_postwar_was_at_war_with_player
                                }
                                any_enemy_country = {
                                    original_tag = event_target:cge_postwar_watched_origin
                                }
                            }
                        }
                        if = {
                            limit = {
                                has_variable = cge_slot_[CGE_SLOT]_runtime_leader
                            }
                            var:cge_slot_[CGE_SLOT]_runtime_leader = {
                                set_character_flag = cge_postwar_tracking_initialized
                                set_character_flag = cge_postwar_war_pending
                            }
                        }
                        set_country_flag = cge_postwar_watch_has_pending_origin
                        set_country_flag = cge_postwar_watch_active
                    }
                    else_if = {
                        limit = {
                            has_variable = cge_slot_[CGE_SLOT]_runtime_leader
                            var:cge_slot_[CGE_SLOT]_runtime_leader = {
                                has_character_flag = cge_postwar_war_pending
                            }
                        }
                        set_country_flag = cge_postwar_watch_has_pending_origin
                        set_country_flag = cge_postwar_watch_active
                    }
                    if = {
                        limit = {
                            check_variable = {
                                var = cge_slot_[CGE_SLOT]_return_pending
                                value = 0
                                compare = greater_than
                            }
                        }
                        set_country_flag = cge_return_stuck_pending_detected
                    }
                }
            }
            CGE_SLOT = "[?cge_slot_arg|.0]"
        }
    }
}
