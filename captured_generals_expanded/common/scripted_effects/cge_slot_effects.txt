cge_slot_sync_patience_decay = {
	meta_effect = {
		text = {
			if = {
				limit = { has_country_flag = cge_has_candidate_[CGE_SLOT] }

				if = {
					limit = {
						NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_executed value = 0 compare = greater_than } }
						NOT = { check_variable = { var = cge_slot_[CGE_SLOT]_recruited value = 0 compare = greater_than } }
					}
					set_variable = { cge_overview_patience_weekly_decay = -1 }
					if = { limit = { check_variable = { var = cge_slot_[CGE_SLOT]_patience value = 75 compare = greater_than } } set_variable = { cge_overview_cooperation_weekly_decay = -0.5 } }
					else_if = { limit = { check_variable = { var = cge_slot_[CGE_SLOT]_patience value = 50 compare = greater_than } } set_variable = { cge_overview_cooperation_weekly_decay = -1 } }
					else_if = { limit = { check_variable = { var = cge_slot_[CGE_SLOT]_patience value = 25 compare = greater_than } } set_variable = { cge_overview_cooperation_weekly_decay = -1.5 } }
					else_if = { limit = { check_variable = { var = cge_slot_[CGE_SLOT]_patience value = 0 compare = greater_than } } set_variable = { cge_overview_cooperation_weekly_decay = -2 } }
					else = { set_variable = { cge_overview_cooperation_weekly_decay = -3 } }
				}

				if = {
					limit = {
						check_variable = { var = cge_slot_[CGE_SLOT]_patience_operation_pending value = 0 compare = greater_than }
						check_variable = { var = cge_slot_[CGE_SLOT]_patience_selected_action value = 1 compare = equals }
					}
					set_variable = { cge_overview_patience_weekly_decay = -2 }
				}

				if = {
					limit = {
						OR = {
							check_variable = { var = cge_slot_[CGE_SLOT]_eased_patience_protection_days_left value = 0 compare = greater_than }
							AND = {
								check_variable = { var = cge_slot_[CGE_SLOT]_cooperation_operation_pending value = 0 compare = greater_than }
								check_variable = { var = cge_slot_[CGE_SLOT]_cooperation_selected_action value = 1 compare = equals }
							}
						}
					}
					set_variable = { cge_overview_patience_weekly_decay = 0 }
				}

				if = {
					limit = {
						check_variable = { var = cge_slot_[CGE_SLOT]_patience_operation_pending value = 0 compare = greater_than }
						check_variable = { var = cge_slot_[CGE_SLOT]_patience_selected_action value = 5 compare = equals }
					}
					set_variable = { cge_overview_patience_weekly_decay = 0 }
					set_variable = { cge_overview_cooperation_weekly_decay = 0 }
				}
			}
		}
		CGE_SLOT = "[?cge_slot_arg|.0]"
	}
}
