cge_restore_missing_recruited_records = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        save_event_target_as = cge_roster_captor
        clr_country_flag = cge_roster_restored_any
        every_army_leader = {
            include_invisible = yes
            limit = {
                has_character_flag = cge_recruited_by_player
                NOT = {
                    has_character_flag = cge_roster_retired
                }
                check_variable = {
                    var = cge_cge_origin_country^num
                    value = 0
                    compare = greater_than
                }
                var:cge_cge_captor_country^0 = {
                    tag = event_target:cge_roster_captor
                }
            }
            save_event_target_as = cge_roster_general
            event_target:cge_roster_captor = {
                clr_country_flag = cge_roster_general_found
                for_loop_effect = {
                    start = 1
                    end = cge_captured_general_count
                    compare = less_than_or_equals
                    value = cge_roster_check_slot
                    meta_effect = {
                        text = {
                            if = {
                                limit = {
                                    has_country_flag = cge_has_candidate_[CGE_SLOT]
                                    check_variable = {
                                        var = cge_slot_[CGE_SLOT]_runtime_leader
                                        value = event_target:cge_roster_general
                                        compare = equals
                                    }
                                }
                                set_country_flag = cge_roster_general_found
                            }
                        }
                        CGE_SLOT = "[?cge_roster_check_slot|.0]"
                    }
                }
                if = {
                    limit = {
                        NOT = {
                            has_country_flag = cge_roster_general_found
                        }
                    }
                    cge_restore_recruited_record = yes
                    set_country_flag = cge_roster_restored_any
                }
            }
        }
        clr_country_flag = cge_roster_general_found
        if = {
            limit = {
                has_country_flag = cge_roster_restored_any
            }
            clr_country_flag = cge_roster_restored_any
            cge_refresh_list_after_change = yes
        }
    }
