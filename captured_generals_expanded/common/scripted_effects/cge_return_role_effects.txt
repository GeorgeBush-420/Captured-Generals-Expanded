cge_return_refresh_saved_role_stats_from_live = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
            is_unit_leader = yes
        }
        set_variable = {
            cge_return_saved_rank = 1
        }
        if = {
            limit = {
                is_field_marshal = yes
            }
            set_variable = {
                cge_return_saved_rank = 2
            }
        }
        set_variable = {
            var = cge_return_saved_level
            value = skill_level
        }
        set_variable = {
            var = cge_return_saved_attack
            value = attack_skill_level
        }
        set_variable = {
            var = cge_return_saved_defense

        }

            var = cge_return_saved_planning
            value = planning_skill_level
        }

            value = logistics_skill_level
        }
    }
}

cge_return_unlock_if_hidden = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        save_event_target_as = cge_return_visibility_character
        if = {
            limit = {
                NOT = {
                    is_unit_leader = yes
                }
            }
            cge_return_rebuild_unrestricted_role = yes
        }
        else_if = {
            limit = {
                has_event_target = cge_return_destination
                event_target:cge_return_destination = {
                    NOT = {
                        is_leader_visible = event_target:cge_return_visibility_character
                    }
                }
            }
            cge_return_refresh_saved_role_stats_from_live = yes
            cge_return_rebuild_unrestricted_role = yes
        }
        else_if = {
            limit = {
                has_event_target = cge_negotiation_partner_country
                event_target:cge_negotiation_partner_country = {
                    NOT = {
                        is_leader_visible = event_target:cge_return_visibility_character
                    }
                }
            }
            cge_return_refresh_saved_role_stats_from_live = yes
            cge_return_rebuild_unrestricted_role = yes
        }
        cge_return_clear_saved_role = yes
    }
}
