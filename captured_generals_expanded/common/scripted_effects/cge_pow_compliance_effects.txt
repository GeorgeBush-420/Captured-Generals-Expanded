cge_pow_compliance_initialize = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        if = {
            limit = {
                NOT = {
                    has_country_flag = cge_pow_compliance_schema_2
                }
            }
            set_country_flag = cge_pow_compliance_schema_2
            set_country_flag = cge_pow_compliance_initialized
            set_country_flag = cge_pow_compliance_migrating
            set_variable = {
                cge_pow_compliance_rows = 0
            }
            for_loop_effect = {
                start = 1
                end = 10
                compare = less_than_or_equals
                value = cge_pow_legacy_row
                meta_effect = {
                    text = {
                        clear_array = cge_pow_legacy_[ROW]_origin
                        if = {
                            limit = {
                                check_variable = {
                                    var = cge_pow_compliance_[ROW]_origin^num
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            add_to_array = {
                                array = cge_pow_legacy_[ROW]_origin
                                value = var:cge_pow_compliance_[ROW]_origin^0
                            }
                        }
                        set_variable = {
                            var = cge_pow_legacy_[ROW]_stacks
                            value = cge_pow_compliance_[ROW]_stacks
                        }
                        clear_array = cge_pow_compliance_[ROW]_origin
                        clear_variable = cge_pow_compliance_[ROW]_stacks
                        clear_variable = cge_pow_compliance_[ROW]_gain
                    }
                    ROW = "[?cge_pow_legacy_row|.0]"
                }
            }
            for_loop_effect = {
                start = 1
                end = 10
                compare = less_than_or_equals
                value = cge_pow_legacy_row
                meta_effect = {
                    text = {
                        if = {
                            limit = {
                                check_variable = {
                                    var = cge_pow_legacy_[ROW]_stacks
                                    value = 0
                                    compare = greater_than
                                }
                                check_variable = {
                                    var = cge_pow_legacy_[ROW]_origin^num
                                    value = 0
                                    compare = greater_than
                                }
                            }
                            var:cge_pow_legacy_[ROW]_origin^0 = {
                                var:original_tag = {
                                    save_event_target_as = cge_pow_compliance_new_origin
                                }
                            }
                            set_variable = {
                                var = cge_pow_compliance_credit_size
                                value = cge_pow_legacy_[ROW]_stacks
                            }
                            cge_pow_compliance_register_origin = yes
                        }
                        clear_array = cge_pow_legacy_[ROW]_origin
                        clear_variable = cge_pow_legacy_[ROW]_stacks
                    }
                    ROW = "[?cge_pow_legacy_row|.0]"
                }
            }
            set_country_flag = cge_pow_compliance_counting_existing
            for_loop_effect = {
                start = 1
                end = cge_captured_general_count
                compare = less_than_or_equals
                value = cge_pow_compliance_legacy_slot
                set_variable = {
                    var = cge_pow_compliance_completion_slot
                    value = cge_pow_compliance_legacy_slot
                }
                cge_pow_compliance_register_slot = yes
            }
            clr_country_flag = cge_pow_compliance_counting_existing
            for_loop_effect = {
                start = 1
                end = cge_pow_compliance_rows
                compare = less_than_or_equals
                value = cge_pow_compliance_row
                meta_effect = {
                    text = {
                        if = {
                            limit = {
                                check_variable = {
                                    var = cge_pow_compliance_[ROW]_observed
                                    value = cge_pow_compliance_[ROW]_stacks
                                    compare = greater_than
                                }
                            }
                            set_variable = {
                                var = cge_pow_compliance_[ROW]_stacks
                                value = cge_pow_compliance_[ROW]_observed
                            }
                        }
                        clear_variable = cge_pow_compliance_[ROW]_observed
                    }
                    ROW = "[?cge_pow_compliance_row|.0]"
                }
            }
            clr_country_flag = cge_pow_compliance_migrating
            clear_variable = cge_pow_compliance_completion_slot
            clear_variable = cge_pow_compliance_credit_size
            cge_pow_compliance_refresh = yes
        }
    }
}
cge_pow_compliance_register_slot = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        meta_effect = {
            text = {
                if = {
                    limit = {
                        has_country_flag = cge_has_candidate_[CGE_SLOT]
                        check_variable = {
                            var = cge_slot_[CGE_SLOT]_pow_collaborator
                            value = 0
                            compare = greater_than
                        }
                        NOT = {
                            check_variable = {
                                var = cge_slot_[CGE_SLOT]_executed
                                value = 0
                                compare = greater_than
                            }
                        }
                        check_variable = {
                            var = cge_slot_[CGE_SLOT]_origin_country^num
                            value = 0
                            compare = greater_than
                        }
                        var:cge_slot_[CGE_SLOT]_runtime_leader = {
                            NOT = {
                                has_character_flag = cge_pow_compliance_registered_v2
                            }
                        }
                    }
                    var:cge_slot_[CGE_SLOT]_runtime_leader = {
                        set_character_flag = cge_pow_compliance_registered_v2
                    }
                    var:cge_slot_[CGE_SLOT]_origin_country^0 = {
                        var:original_tag = {
                            save_event_target_as = cge_pow_compliance_new_origin
                        }
                    }
                    set_variable = {
                        cge_pow_compliance_credit_size = 1
                    }
                    cge_pow_compliance_register_origin = yes
                    clear_variable = cge_pow_compliance_credit_size
                    if = {
                        limit = {
                            NOT = {
                                has_country_flag = cge_pow_compliance_migrating
                            }
                        }
                        cge_pow_compliance_refresh = yes
                    }
                }
            }
            CGE_SLOT = "[?cge_pow_compliance_completion_slot|.0]"
        }
    }
}
cge_pow_compliance_register_origin = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        if = {
            limit = {
                any_enemy_country = {
                    original_tag = event_target:cge_pow_compliance_new_origin
                }
            }
            set_variable = {
                cge_pow_compliance_matching_row = 0
            }
            set_variable = {
                cge_pow_compliance_empty_row = 0
            }
            for_loop_effect = {
                start = 1
                end = cge_pow_compliance_rows
                compare = less_than_or_equals
                value = cge_pow_compliance_find_index
                meta_effect = {
                    text = {
                        if = {
                            limit = {
                                check_variable = {
                                    var = cge_pow_compliance_[ROW]_origin^num
                                    value = 0
                                    compare = greater_than
                                }
                                var:cge_pow_compliance_[ROW]_origin^0 = {
                                    tag = event_target:cge_pow_compliance_new_origin
                                }
                            }
                            set_variable = {
                                cge_pow_compliance_matching_row = [ROW]
                            }
                        }
                        else_if = {
                            limit = {
                                NOT = {
                                    check_variable = {
                                        var = cge_pow_compliance_[ROW]_origin^num
                                        value = 0
                                        compare = greater_than
                                    }
                                }
                                check_variable = {
                                    var = cge_pow_compliance_empty_row
                                    value = 0
                                    compare = equals
                                }
                            }
                            set_variable = {
                                cge_pow_compliance_empty_row = [ROW]
                            }
                        }
                    }
                    ROW = "[?cge_pow_compliance_find_index|.0]"
                }
            }
            if = {
                limit = {
                    check_variable = {
                        var = cge_pow_compliance_matching_row
                        value = 0
                        compare = equals
                    }
                }
                if = {
                    limit = {
                        check_variable = {
                            var = cge_pow_compliance_empty_row
                            value = 0
                            compare = greater_than
                        }
                    }
                    set_variable = {
                        var = cge_pow_compliance_matching_row
                        value = cge_pow_compliance_empty_row
                    }
                }
                else = {
                    add_to_variable = {
                        cge_pow_compliance_rows = 1
                    }
                    set_variable = {
                        var = cge_pow_compliance_matching_row
                        value = cge_pow_compliance_rows
                    }
                }
            }
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            NOT = {
                                check_variable = {
                                    var = cge_pow_compliance_[ROW]_origin^num
                                    value = 0
                                    compare = greater_than
                                }
                            }
                        }
                        clear_array = cge_pow_compliance_[ROW]_origin
                        add_to_array = {
                            array = cge_pow_compliance_[ROW]_origin
                            value = event_target:cge_pow_compliance_new_origin
                        }
                        set_variable = {
                            cge_pow_compliance_[ROW]_stacks = 0
                        }
                    }
                    if = {
                        limit = {
                            has_country_flag = cge_pow_compliance_counting_existing
                        }
                        add_to_variable = {
                            cge_pow_compliance_[ROW]_observed = 1
                        }
                    }
                    else = {
                        add_to_variable = {
                            var = cge_pow_compliance_[ROW]_stacks
                            value = cge_pow_compliance_credit_size
                        }
                        clamp_variable = {
                            var = cge_pow_compliance_[ROW]_stacks
                            min = 0
                            max = 10
                        }
                    }
                }
                ROW = "[?cge_pow_compliance_matching_row|.0]"
            }
            clear_variable = cge_pow_compliance_matching_row
            clear_variable = cge_pow_compliance_empty_row
        }
    }
}
cge_pow_compliance_refresh = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        set_variable = {
            cge_pow_compliance_total = 0
        }
        set_variable = {
            cge_pow_compliance_overflow_countries = 0
        }
        for_loop_effect = {
            start = 1
            end = cge_pow_compliance_rows
            compare = less_than_or_equals
            value = cge_pow_compliance_row
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            check_variable = {
                                var = cge_pow_compliance_[ROW]_stacks
                                value = 0
                                compare = greater_than
                            }
                            check_variable = {
                                var = cge_pow_compliance_[ROW]_origin^num
                                value = 0
                                compare = greater_than
                            }
                            any_enemy_country = {
                                original_tag = var:PREV.cge_pow_compliance_[ROW]_origin^0
                            }
                        }
                        clamp_variable = {
                            var = cge_pow_compliance_[ROW]_stacks
                            min = 0
                            max = 10
                        }
                        set_variable = {
                            var = cge_pow_compliance_[ROW]_gain
                            value = cge_pow_compliance_[ROW]_stacks
                        }
                        multiply_variable = {
                            cge_pow_compliance_[ROW]_gain = 0.1
                        }
                        add_to_variable = {
                            var = cge_pow_compliance_total
                            value = cge_pow_compliance_[ROW]_stacks
                        }
                        if = {
                            limit = {
                                check_variable = {
                                    var = cge_pow_compliance_row
                                    value = 128
                                    compare = greater_than
                                }
                            }
                            add_to_variable = {
                                cge_pow_compliance_overflow_countries = 1
                            }
                        }
                    }
                    else = {
                        clear_array = cge_pow_compliance_[ROW]_origin
                        clear_variable = cge_pow_compliance_[ROW]_stacks
                        clear_variable = cge_pow_compliance_[ROW]_gain
                    }
                }
                ROW = "[?cge_pow_compliance_row|.0]"
            }
        }
        if = {
            limit = {
                check_variable = {
                    var = cge_pow_compliance_total
                    value = 0
                    compare = greater_than
                }
            }
            set_country_flag = cge_pow_compliance_active
            if = {
                limit = {
                    NOT = {
                        has_idea = cge_pow_collaboration_network
                    }
                }
                add_ideas = cge_pow_collaboration_network
            }
        }
        else = {
            clr_country_flag = cge_pow_compliance_active
            if = {
                limit = {
                    has_idea = cge_pow_collaboration_network
                }
                remove_ideas = cge_pow_collaboration_network
            }
        }
        cge_pow_compliance_sync_states = yes
    }
}
cge_pow_compliance_daily_tick = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        cge_pow_compliance_refresh = yes
    }
}
cge_pow_compliance_sync_states = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        save_event_target_as = cge_pow_compliance_captor
        every_state = {
            limit = {
                has_state_flag = cge_pow_compliance_managed_state
                var:cge_pow_compliance_beneficiary = {
                    tag = event_target:cge_pow_compliance_captor
                }
            }
            set_variable = {
                cge_pow_compliance_state_gain = 0
            }
        }
        for_loop_effect = {
            start = 1
            end = cge_pow_compliance_rows
            compare = less_than_or_equals
            value = cge_pow_compliance_apply_index
            meta_effect = {
                text = {
                    if = {
                        limit = {
                            check_variable = {
                                var = cge_pow_compliance_[ROW]_stacks
                                value = 0
                                compare = greater_than
                            }
                        }
                        every_controlled_state = {
                            limit = {
                                has_resistance = yes
                                OCCUPIED = {
                                    original_tag = var:PREV.PREV.cge_pow_compliance_[ROW]_origin^0
                                }
                            }
                            set_variable = {
                                var = cge_pow_compliance_state_gain
                                value = PREV.cge_pow_compliance_[ROW]_gain
                            }
                            set_variable = {
                                var = cge_pow_compliance_state_origin
                                value = var:PREV.cge_pow_compliance_[ROW]_origin^0
                            }
                            set_variable = {
                                var = cge_pow_compliance_beneficiary
                                value = event_target:cge_pow_compliance_captor
                            }
                            set_state_flag = cge_pow_compliance_managed_state
                            if = {
                                limit = {
                                    NOT = {
                                        has_dynamic_modifier = {
                                            modifier = cge_pow_collaboration_growth
                                        }
                                    }
                                }
                                add_dynamic_modifier = {
                                    modifier = cge_pow_collaboration_growth
                                }
                            }
                            force_update_dynamic_modifier = yes
                        }
                    }
                }
                ROW = "[?cge_pow_compliance_apply_index|.0]"
            }
        }
        every_state = {
            limit = {
                has_state_flag = cge_pow_compliance_managed_state
                var:cge_pow_compliance_beneficiary = {
                    tag = event_target:cge_pow_compliance_captor
                }
                NOT = {
                    cge_pow_compliance_state_active = yes
                }
            }
            cge_pow_compliance_clear_state = yes
        }
    }
}
cge_pow_compliance_clear_state = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        if = {
            limit = {
                has_dynamic_modifier = {
                    modifier = cge_pow_collaboration_growth
                }
            }
            remove_dynamic_modifier = {
                modifier = cge_pow_collaboration_growth
            }
        }
        clr_state_flag = cge_pow_compliance_managed_state
        clear_variable = cge_pow_compliance_state_gain
        clear_variable = cge_pow_compliance_state_origin
        clear_variable = cge_pow_compliance_beneficiary
    }
}
