cge_snapshot_live_stats_on_from = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        event_target:cge_capture_stats_captor = {
            set_variable = {
                var = cge_cap_attack_bonus
                value = cge_cap_attack
            }
            multiply_variable = {
                var = cge_cap_attack_bonus
                value = 2.5
            }
            set_variable = {
                var = cge_cap_defense_bonus
                value = cge_cap_defense
            }
            multiply_variable = {
                var = cge_cap_defense_bonus
                value = 2.5
            }
            set_variable = {
                var = cge_cap_planning_bonus
                value = cge_cap_planning
            }
            multiply_variable = {
                var = cge_cap_planning_bonus
                value = 2.5
            }
            set_variable = {
                var = cge_cap_logistics_bonus
                value = cge_cap_logistics
            }
            multiply_variable = {
                var = cge_cap_logistics_bonus
                value = 2.5
            }
        }
        if = {
            limit = {
                is_field_marshal = yes
            }
            event_target:cge_capture_stats_captor = {
                set_variable = {
                    cge_cap_rank = 2
                }
                set_variable = {
                    cge_cap_capacity = 5
                }
            }
        }
        else = {
            event_target:cge_capture_stats_captor = {
                set_variable = {
                    cge_cap_rank = 1
                }
                set_variable = {
                    cge_cap_capacity = 24
                }
            }
        }
    }
}
cge_identify_character_on_from = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        cge_identify_macarthur = yes
        if = {
            limit = {
                OR = {
                    NOT = {
                        has_variable = cge_registry_character_key
                    }
                    NOT = {
                        has_variable = cge_registry_name_order
                    }
                }
            }
            cge_identify_character_on_from_base = yes
        }
        if = {
            limit = {
                has_variable = cge_registry_character_key
            }
            event_target:cge_capture_stats_captor = {
                set_variable = {
                    var = cge_cap_char_key
                    value = PREV.cge_registry_character_key
                }
                set_variable = {
                    cge_cap_name_order = 1000000
                }
                if = {
                    limit = {
                        PREV = {
                            has_variable = cge_registry_name_order
                        }
                    }
                    set_variable = {
                        var = cge_cap_name_order
                        value = PREV.cge_registry_name_order
                    }
                }
            }
        }
    }
}

cge_append_capture_trait_id = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        if = {
            limit = {
                check_variable = {
                    var = cge_cap_trait_count
                    value = 8
                    compare = less_than
                }
            }
            meta_effect = {
                text = {
                    set_variable = {
                        cge_tmp_trait_[CGE_POS] = [CGE_ID]
                    }
                }
                CGE_POS = "[?cge_cap_trait_pos|.0]"
                CGE_ID = "[?cge_cap_trait_id|.0]"
            }
            add_to_variable = {
                cge_cap_trait_count = 1
            }
        }
        clear_variable = cge_cap_trait_pos
        clear_variable = cge_cap_trait_id
    }

cge_prepare_capture_snapshot_on_from = {
    if = {
        limit = {
            NOT = {
                has_global_flag = cge_logic_disabled
            }
        }
        FROM = {
            cge_clear_capture_temp_data = yes
            save_event_target_as = cge_capture_stats_captor
            clr_country_flag = cge_capture_actual_leader_found
        }
        OWNER = {
            save_event_target_as = cge_capture_origin_country
        }
        FROM = {
            var:original_tag = {
                save_event_target_as = cge_capture_captor_original_tag
            }
        }
        OWNER = {
            if = {
                limit = {
                    OR = {
                        civilwar_target = event_target:cge_capture_stats_captor
                        AND = {
                            has_civil_war = yes
                            event_target:cge_capture_stats_captor = {
                                has_civil_war = yes
                            }
                            original_tag = event_target:cge_capture_captor_original_tag
                        }
                    }
                }
                event_target:cge_capture_stats_captor = {
                    set_variable = {
                        cge_cap_domestic_civil_war = 1
                    }
                }
            }
        }
        cge_snapshot_live_traits_on_from = yes
        set_unit_leader_flag = cge_capture_resolve_marker
        OWNER = {
            random_army_leader = {
                include_invisible = yes
                limit = {
                    has_unit_leader_flag = cge_capture_resolve_marker
                }
                clr_unit_leader_flag = cge_capture_resolve_marker
                save_event_target_as = cge_capture_actual_leader
                cge_identify_character_on_from = yes
                if = {
                    limit = {
                        event_target:cge_capture_stats_captor = {
                            check_variable = {
                                var = cge_cap_char_key
                                value = 1
                                compare = less_than
                            }
                        }
                        NOT = {
                            has_variable = cge_runtime_portrait_key
                        }
                    }
                    cge_assign_owner_portrait = yes
                }
                if = {
                    limit = {
                        has_variable = cge_runtime_portrait_key
                    }
                    event_target:cge_capture_stats_captor = {
                        set_variable = {
                            var = cge_cap_portrait_key
                            value = PREV.cge_runtime_portrait_key
                        }
                    }
                }
                cge_capture_character_stats = yes
                cge_snapshot_live_stats_on_from = yes
                if = {
                    limit = {
                        has_variable = cge_capture_site_state
                    }
                    event_target:cge_capture_stats_captor = {
                        set_variable = {
                            var = cge_cap_state
                            value = PREV.cge_capture_site_state
                        }
                    }
                }
                if = {
                    limit = {
                        event_target:cge_capture_stats_captor = {
                            check_variable = {
                                var = cge_cap_char_key
                                value = 1
                                compare = less_than
                            }
                        }
                    }
                    if = {
                        limit = {
                            check_variable = {
                                var = cge_runtime_prisoner_id
                                value = 1
                                compare = less_than
                            }
                        }
                        event_target:cge_capture_stats_captor = {
                            add_to_variable = {
                                cge_runtime_prisoner_sequence = 1
                            }
                        }
                        set_variable = {
                            var = cge_runtime_prisoner_id
                            value = event_target:cge_capture_stats_captor.cge_runtime_prisoner_sequence
                        }
                    }
                    event_target:cge_capture_stats_captor = {
                        set_variable = {
                            var = cge_cap_prisoner_id
                            value = PREV.cge_runtime_prisoner_id
                        }
                        set_variable = {
                            cge_cap_name_order = 1000000
                        }
                        add_to_variable = {
                            var = cge_cap_name_order
                            value = cge_cap_prisoner_id
                        }
                    }
                }
                event_target:cge_capture_stats_captor = {
                    set_country_flag = cge_capture_actual_leader_found
                }
            }
        }
        clr_unit_leader_flag = cge_capture_resolve_marker
        FROM = {
            if = {
                limit = {
                    NOT = {
                        has_country_flag = cge_capture_actual_leader_found
                    }
                }
            }
        }
    }
